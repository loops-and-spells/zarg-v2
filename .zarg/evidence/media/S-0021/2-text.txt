+export function assign(
+  first: string | FamilyChores,
+  childId?: string,
+  chore?: string,
+): ReadonlyArray<string> | FamilyChores {
+  if (typeof first === "string") {
+    familyChores.push(first)
+    return listChores()
+  }
+  return {
+    parent: [...first.parent, chore as string],
+    children: {
+      ...first.children,
+      [childId as string]: [...(first.children[childId as string] ?? []), chore as string],
+    },
+  }
+}

+export const remove = (chores: FamilyChores, chore: string): FamilyChores => ({
+  parent: chores.parent.filter((existing) => existing !== chore),
+  children: Object.fromEntries(
+    Object.entries(chores.children).map(([childId, list]) => [
+      childId,
+      list.filter((existing) => existing !== chore),
+    ]),
+  ),
+})
